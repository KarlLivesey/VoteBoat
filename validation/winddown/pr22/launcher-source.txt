import hashlib,json,os,pathlib,signal,subprocess,sys,time
os.setpriority(os.PRIO_PROCESS,0,19)
os.sched_setaffinity(0,{0,1,2,3})
root=pathlib.Path.cwd()
report=pathlib.Path('/home/karllivesey/Documents/ChatGPT/VoteBoat/target/claude/research-r4/pr22-final-main')
label,*cmd=sys.argv[1:]
env=os.environ.copy()
env.update(CARGO_BUILD_JOBS='4',CARGO_INCREMENTAL='0',CARGO_PROFILE_DEV_DEBUG='0',CARGO_PROFILE_TEST_DEBUG='0',CARGO_TARGET_DIR=str(root/'target'),RUST_TEST_THREADS='4',PYTHONDONTWRITEBYTECODE='1')
for k in ('RUSTFLAGS','CARGO_ENCODED_RUSTFLAGS'): env.pop(k,None)
guard='/home/karllivesey/Documents/ChatGPT/VoteBoat/target/claude/tools/diskguard.sh'
def guarded(suffix):
 with (report/f'{label}.{suffix}.guard').open('w') as f:
  rc=subprocess.run([guard],stdout=f,stderr=subprocess.STDOUT,env=env).returncode
 if rc: raise SystemExit(f'guard failed {rc}')
guarded('before')
os.setpriority(os.PRIO_PROCESS,0,19)
os.sched_setaffinity(0,{0,1,2,3})
record={'label':label,'command':cmd,'cwd':str(root),'env':{k:env[k] for k in ('CARGO_BUILD_JOBS','CARGO_INCREMENTAL','CARGO_PROFILE_DEV_DEBUG','CARGO_PROFILE_TEST_DEBUG','CARGO_TARGET_DIR','RUST_TEST_THREADS')},'controller_pid':os.getpid(),'controller_nice':os.getpriority(os.PRIO_PROCESS,0),'controller_affinity':sorted(os.sched_getaffinity(0)),'samples':0,'task_records':0,'violations':[]}
start=time.monotonic()
with (report/f'{label}.log').open('w') as log:
 def child_resources():
  os.setpriority(os.PRIO_PROCESS,0,19)
  os.sched_setaffinity(0,{0,1,2,3})
 p=subprocess.Popen(cmd,env=env,stdout=log,stderr=subprocess.STDOUT,start_new_session=True,preexec_fn=child_resources)
 os.setpriority(os.PRIO_PROCESS,p.pid,19)
 record['child_launch_nice']=os.getpriority(os.PRIO_PROCESS,p.pid)
 record['child_launch_affinity']=sorted(os.sched_getaffinity(p.pid))
 try:
  while p.poll() is None:
   parents={}
   for proc in pathlib.Path('/proc').glob('[0-9]*'):
    try: parents[int(proc.name)]=int((proc/'stat').read_text().rsplit(')',1)[1].split()[1])
    except (OSError,ValueError,IndexError): pass
   owned={os.getpid()}
   while True:
    added={pid for pid,parent in parents.items() if parent in owned}-owned
    if not added: break
    owned.update(added)
   for pid in owned:
    for task in pathlib.Path(f'/proc/{pid}/task').glob('[0-9]*'):
     try:
      tid=int(task.name); nice=os.getpriority(os.PRIO_PROCESS,tid); cpus=sorted(os.sched_getaffinity(tid))
      record['task_records']+=1
      if nice!=19 or not set(cpus)<={0,1,2,3}:
       record['violations'].append({'pid':pid,'tid':tid,'nice':nice,'cpus':cpus})
     except (ProcessLookupError,FileNotFoundError): pass
   record['samples']+=1
   size=int(subprocess.check_output(['du','-sb',str(root/'target')],text=True).split()[0])
   record['max_target_bytes']=max(size,record.get('max_target_bytes',0))
   if size>10*1024**3 or record['violations'] or time.monotonic()-start>900:
    os.killpg(p.pid,signal.SIGTERM)
    raise RuntimeError('resource/deadline violation')
   time.sleep(1)
  record['rc']=p.wait()
 finally:
  if p.poll() is None:
   os.killpg(p.pid,signal.SIGKILL); p.wait()
  record.setdefault('rc',p.returncode)
  record['seconds']=round(time.monotonic()-start,3)
  record['log_sha256']=hashlib.sha256((report/f'{label}.log').read_bytes()).hexdigest()
  (report/f'{label}.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps(record))
guarded('after')
sys.exit(record['rc'])
